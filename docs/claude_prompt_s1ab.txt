from __future__ import annotations
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional
from core.state import QuantumState


class EpistemicDecision(str, Enum):
    EXECUTE_DIRECTLY = "EXECUTE_DIRECTLY"
    NEED_KNOWLEDGE_FORAGING = "NEED_KNOWLEDGE_FORAGING"
    DELEGATE_MODEL = "DELEGATE_MODEL"


@dataclass
class GateContext:
    task_description: str
    confidence_threshold: float = 0.7
    unknown_entities: list[str] = field(default_factory=list)
    metadata: dict = field(default_factory=dict)  # type: ignore[type-arg]


@dataclass
class GateResult:
    decision: EpistemicDecision
    confidence: float
    unknown_entities: list[str]
    rationale: str
    suggested_modality: Optional[str] = None
    suggested_model: Optional[str] = None


class EpistemicGate:
    """Blocking epistemic assessment gate between evolve and evaluate.

    V5.0 Architecture:
        encode -> evolve -> [EPISTEMIC GATE] -> evaluate -> decode

    The gate emits one of three decisions:
        EXECUTE_DIRECTLY: high confidence, no unknowns - proceed
        NEED_KNOWLEDGE_FORAGING: unknown entities detected - acquire knowledge first
        DELEGATE_MODEL: low confidence, no foraging path - route to specialist
    """

    def __init__(self, confidence_threshold: float = 0.7) -> None:
        self.confidence_threshold = confidence_threshold

    def assess(self, state: QuantumState, context: GateContext) -> GateResult:
        """Assess epistemic state and return routing decision.

        This is the BLOCKING step replacing parallel funnel execution.
        Must be called between evolve() and evaluate() in the orchestrator.
        """
        confidence = self._compute_confidence(state)
        unknown_entities = self._detect_unknown_entities(state, context)

        if confidence >= self.confidence_threshold and not unknown_entities:
            return GateResult(
                decision=EpistemicDecision.EXECUTE_DIRECTLY,
                confidence=confidence,
                unknown_entities=[],
                rationale=f"Confidence {confidence:.2f} >= threshold {self.confidence_threshold:.2f}",
            )

        if unknown_entities:
            return GateResult(
                decision=EpistemicDecision.NEED_KNOWLEDGE_FORAGING,
                confidence=confidence,
                unknown_entities=unknown_entities,
                rationale=f"Unknown entities detected: {unknown_entities}",
                suggested_modality="text",
            )

        return GateResult(
            decision=EpistemicDecision.DELEGATE_MODEL,
            confidence=confidence,
            unknown_entities=[],
            rationale=f"Confidence {confidence:.2f} below threshold, no foraging target",
            suggested_model="general",
        )

    def _compute_confidence(self, state: QuantumState) -> float:
        """Derive confidence from state norm, clamped to [0.0, 1.0]."""
        try:
            norm = float(state.norm())
            return min(1.0, max(0.0, norm))
        except Exception:
            return 0.0

    def _detect_unknown_entities(
        self, state: QuantumState, context: GateContext
    ) -> list[str]:
        """Return unknown entities from context (populated by encoder in V5.0)."""
        return list(context.unknown_entities)
